`src/core/sovereigntyMonitor.ts`

export type SovereigntyEvent = {
  type: 'external' | 'internal';
  provider?: string;
  engine?: string;
  success: boolean;
  latencyMs?: number;
  timestamp: string;
};

const events: SovereigntyEvent[] = [];

export const sovereigntyMonitor = {
  record(event: SovereigntyEvent) {
    events.push(event);
  },
  summary() {
    const external = events.filter(e => e.type === 'external').length;
    const internal = events.filter(e => e.type === 'internal').length;
    return { external, internal, total: events.length };
  },
  all() {
    return events;
  },
};

`src/providers/httpClient.ts`

import fetch, { RequestInit } from 'node-fetch';
import { sovereigntyMonitor } from '../core/sovereigntyMonitor';

export async function safeFetch(
  provider: string,
  url: string,
  options: RequestInit = {},
) {
  const start = Date.now();
  try {
    const res = await fetch(url, options);
    sovereigntyMonitor.record({
      type: 'external',
      provider,
      success: res.ok,
      latencyMs: Date.now() - start,
      timestamp: new Date().toISOString(),
    });
    return res;
  } catch (err) {
    sovereigntyMonitor.record({
      type: 'external',
      provider,
      success: false,
      latencyMs: Date.now() - start,
      timestamp: new Date().toISOString(),
    });
    throw err;
  }
}

import { runThroughSovereignEngine } from '../engines/sovereignEngineRouter';

export type EvalCase = {
  id: string;
  description: string;
  domain: SovereignRequest['domain'];
  query: string;
  mustContain: string[];
};

const evalCases: EvalCase[] = [
  {
    id: 'quantum_superposition_basic',
    description: 'Basic quantum superposition understanding',
    domain: 'knowledge',
    query: 'quantum superposition',
    mustContain: ['superposition', 'quantum'],
  },
];

export async function runEvaluation() {
  const results: { id: string; passed: boolean }[] = [];
  for (const c of evalCases) {
    const res = await runThroughSovereignEngine({
      task: c.query,
      domain: c.domain,
    });
    const text = JSON.stringify(res).toLowerCase();
    const passed = c.mustContain.every(s => text.includes(s.toLowerCase()));
    results.push({ id: c.id, passed });
  }
  return results;
}
