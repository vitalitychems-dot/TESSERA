// src/core/sovereigntyMonitor.ts
export type SovereigntyEvent = {
  type: 'external_call' | 'internal_call';
  provider?: string;
  engine?: string;
  success: boolean;
  latencyMs?: number;
  timestamp: string;
};

const events: SovereigntyEvent[] = [];

export const sovereigntyMonitor = {
  record(event: SovereigntyEvent) {
    events.push(event);
  },
  summary() {
    const external = events.filter(e => e.type === 'external_call').length;
    const internal = events.filter(e => e.type === 'internal_call').length;
    return { external, internal, total: events.length };
  },
};

3. Step 2 — Route all external calls through sovereign engines

Goal: Nothing hits the network except through your sovereign provider layer.

Actions:

• Create a safe HTTP client and provider registry.


// src/providers/httpClient.ts
import fetch, { RequestInit } from 'node-fetch';
import { logger } from '../core/logger';
import { sovereigntyMonitor } from '../core/sovereigntyMonitor';

export async function safeFetch(
  provider: string,
  url: string,
  options: RequestInit = {},
): Promise<Response> {
  const start = Date.now();
  try {
    const res = await fetch(url, { ...options, timeout: 15000 as any });
    const latencyMs = Date.now() - start;
    sovereigntyMonitor.record({
      type: 'external_call',
      provider,
      success: res.ok,
      latencyMs,
      timestamp: new Date().toISOString(),
    });
    if (!res.ok) {
      logger.error('Provider error', { provider, status: res.status, url });
    }
    return res;
  } catch (err) {
    const latencyMs = Date.now() - start;
    sovereigntyMonitor.record({
      type: 'external_call',
      provider,
      success: false,
      latencyMs,
      timestamp: new Date().toISOString(),
    });
    logger.error('Provider fetch failed', { provider, url, err });
    throw err;
  }
}

// src/reasoning/evaluator.ts
import { runThroughSovereignEngine } from '../engines/sovereignEngineRouter';

export type EvalCase = {
  id: string;
  description: string;
  domain: 'knowledge';
  query: string;
  mustContain: string[]; // substrings that should appear in answer
};

const evalCases: EvalCase[] = [
  {
    id: 'wiki_quantum_superposition',
    description: 'Basic quantum superposition knowledge',
    domain: 'knowledge',
    query: 'quantum superposition',
    mustContain: ['superposition', 'quantum'],
  },
];

export async function runEvaluation() {
  const results = [];
  for (const c of evalCases) {
    const response = await runThroughSovereignEngine({ task: c.query, domain: c.domain });
    const text = JSON.stringify(response.result).toLowerCase();
    const passed = c.mustContain.every(s => text.includes(s.toLowerCase()));
    results.push({ id: c.id, passed });
  }
  return results;
}
