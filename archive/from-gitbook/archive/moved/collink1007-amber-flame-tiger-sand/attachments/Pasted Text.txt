from abc import ABC, abstractmethod
from typing import Any, Dict, List

class BaseProvider(ABC):
    def __init__(self, name: str, config: Dict[str, Any]):
        self.name = name
        self.config = config

    @abstractmethod
    def infer(self, prompt: str, **kwargs) -> Dict[str, Any]:
        """Return dict with keys: text, raw, latency, status, metadata."""
        raise NotImplementedError

class ProviderRegistry:
    def __init__(self):
        self._providers: Dict[str, BaseProvider] = {}

    def register(self, name: str, provider: BaseProvider):
        self._providers[name] = provider

    def get(self, name: str) -> BaseProvider:
        return self._providers[name]

    def all(self) -> List[BaseProvider]:
        return list(self._providers.values())

    def init_from_config(self, cfg: Dict[str, Any]):
        from .local_llm import LocalLLMProvider
        from .external_llm import ExternalLLMProvider

        for name, pcfg in cfg.get("providers", {}).items():
            ptype = pcfg.get("type")
            if ptype == "local_llm":
                self.register(name, LocalLLMProvider(name, pcfg))
            elif ptype == "external_llm":
                self.register(name, ExternalLLMProvider(name, pcfg))

import time
import requests
from typing import Any, Dict
from .base_provider import BaseProvider

class LocalLLMProvider(BaseProvider):
    def infer(self, prompt: str, **kwargs) -> Dict[str, Any]:
        start = time.time()
        url = self.config["base_url"]
        payload = {
            "prompt": prompt,
            "max_tokens": kwargs.get("max_tokens", 512),
        }
        try:
            resp = requests.post(url, json=payload, timeout=kwargs.get("timeout", 30))
            resp.raise_for_status()
            data = resp.json()
            text = data.get("text") or data.get("choices", [{}])[0].get("text", "")
            return {
                "text": text,
                "raw": data,
                "latency": time.time() - start,
                "status": "ok",
                "metadata": {"provider": self.name},
            }
        except Exception as e:
            return {
                "text": "",
                "raw": {"error": str(e)},
                "latency": time.time() - start,
                "status": "error",
                "metadata": {"provider": self.name},
            }

import time
import requests
from typing import Any, Dict
from .base_provider import BaseProvider

class ExternalLLMProvider(BaseProvider):
    def infer(self, prompt: str, **kwargs) -> Dict[str, Any]:
        start = time.time()
        url = self.config["base_url"]
        headers = {}
        if "api_key" in self.config:
            headers["Authorization"] = f"Bearer {self.config['api_key']}"
        payload = {
            "model": self.config.get("model", "default"),
            "prompt": prompt,
            "max_tokens": kwargs.get("max_tokens", 512),
        }
        try:
            resp = requests.post(url, json=payload, headers=headers, timeout=kwargs.get("timeout", 30))
            resp.raise_for_status()

class WebScraper:
    def __init__(self, config: Dict[str, Any]):
        self.config = config

    def fetch(self, url: str) -> Dict[str, Any]:
        headers = {"User-Agent": self.config.get("user_agent", "TesseraScraper/1.0")}
        resp = requests.get(url, headers=headers, timeout=self.config.get("timeout", 15))
        resp.raise_for_status()
        html = resp.text
        text = self._html_to_text(html)
        doc_id = hashlib.sha256(url.encode("utf-8")).hexdigest()
        return {
            "id": doc_id,
            "url": url,
            "content": text,
            "timestamp": time.time(),
            "type": "web",
            "tags": [],
        }

    def _html_to_text(self, html: str) -> str:
        soup = BeautifulSoup(html, "html.parser")
        for script in soup(["script", "style"]):
            script.decompose()
        return soup.get_text(separator="\n", strip=True)

import fetch from "node-fetch";
import { BaseProvider } from "./baseProvider";
import { ProviderResponse } from "../core/types";

export class LocalLLMProvider extends BaseProvider {
  async infer(prompt: string, options?: { maxTokens?: number; timeoutMs?: number }): Promise<ProviderResponse> {
    const start = performance.now();
    const url = this.config.baseUrl!;
    const body = {
      prompt,
      max_tokens: options?.maxTokens ?? 512
    };

    try {
      const res = await fetch(url, {
        method: "POST",
        body: JSON.stringify(body),
        headers: { "Content-Type": "application/json" }
      });

      const json = (await res.json()) as any;
      const text =
        json.text ??
        json.choices?.[0]?.text ??
        "";

      return {
        text,
        raw: json,
        latencyMs: performance.now() - start,
        status: "ok",
        metadata: { provider: this.name, local: true }
      };
    } catch (err) {
      return {
        text: "",
        raw: { error: String(err) },
        latencyMs: performance.now() - start,
        status: "error",
        metadata: { provider: this.name, local: true }
      };
    }
  }
}

import fetch from "node-fetch";
import { BaseProvider } from "./baseProvider";
import { ProviderResponse } from "../core/types";

export class ExternalLLMProvider extends BaseProvider {
  async infer(prompt: string, options?: { maxTokens?: number; timeoutMs?: number }): Promise<ProviderResponse> {
    const start = performance.now();
    const url = this.config.baseUrl!;
    const apiKey = this.config.apiKeyEnv ? process.env[this.config.apiKeyEnv] : undefined;

    const body = {
      model: this.config.model ?? "default",
      prompt,
      max_tokens: options?.maxTokens ?? 512
    };

    try {
      const res = await fetch(url, {
        method: "POST",
        body: JSON.stringify(body),
        headers: {
          "Content-Type": "application/json",
          ...(apiKey ? { Authorization: `Bearer ${apiKey}` } : {})
        }
      });

      const json = (await res.json()) as any;
      const text =
        json.text ??
        json.choices?.[0]?.text ??
        "";

      return {
        text,
        raw: json,
        latencyMs: performance.now() - start,
        status: "ok",
        metadata: { provider: this.name, local: false }
      };
    } catch (err) {
      return {
        text: "",
        raw: { error: String(err) },
        latencyMs: performance.now() - start,
        status: "error",
        metadata: { provider: this.name, local: false }
      };
    }
  }
}

src/providers/registry.ts
ts

import { BaseProvider } from "./baseProvider";
import { LocalLLMProvider } from "./localLLMProvider";
import { ExternalLLMProvider } from "./externalLLMProvider";
import { ProviderConfig } from "../core/types";

export class ProviderRegistry {
  private providers = new Map<string, BaseProvider>();

  register(config: ProviderConfig): void {
    let provider: BaseProvider;

    switch (config.type) {
      case "local_llm":
        provider = new LocalLLMProvider(config);
        break;
      case "external_llm":
        provider = new ExternalLLMProvider(config);
        break;
      default:
        throw new Error(`Unknown provider type: ${config.type}`);
    }

    this.providers.set(config.name, provider);
  }

  get(name: string): BaseProvider {
    const p = this.providers.get(name);
    if (!p) throw new Error(`Provider not found: ${name}`);
    return p;
  }

  all(): BaseProvider[] {
    return [...this.providers.values()];
  }
}

import { SovereigntyConfig, SovereigntyMetrics, EvaluationReport } from "../core/types";

export class SovereigntyMonitor {
  private metrics: SovereigntyMetrics = {
    internalRatio: 0,
    externalRatio: 1,
    lastUpdated: Date.now()
  };

  constructor(private readonly config: SovereigntyConfig) {}

  updateFromUsage(internalCalls: number, externalCalls: number): void {
    const total = internalCalls + externalCalls || 1;
    this.metrics.internalRatio = internalCalls / total;
    this.metrics.externalRatio = externalCalls / total;
    this.metrics.lastUpdated = Date.now();
  }

  updateFromEval(_report: EvaluationReport): void {
    // hook for future: adjust confidence based on eval results
    this.metrics.lastUpdated = Date.now();
  }

  isReadyToDisconnect(): boolean {
    return this.metrics.internalRatio >= this.config.disconnectThreshold;
  }

  getMetrics(): SovereigntyMetrics {
    return this.metrics;
  }
}
